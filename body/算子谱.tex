\section{线性算子谱}\label{xxszp}
回顾有限维的情况，设$X$为$n$维的线性空间，$A: \ X \to X$是线性算子，对$\forall \lambda \in \mathbb{C}$考虑方程$(\lambda \mathbbm{I}-A)x=0$，也即$Ax=\lambda x$，有两种情况：
\begin{itemize}
\item 1. 只有零解，算子$\lambda \mathbbm{I}-A$为单射（$\ker(\lambda \mathbbm{I}-A)=\{0\}$且线性），因此$\lambda \mathbbm{I}-A$可逆[定理\ref{theorem:djm}]，也可称为正则，其行列式$\det(\lambda \mathbbm{I}-A)\neq0$，此时称$\lambda$为$A$的正则值；
\item 2. 有非零解，算子$\lambda \mathbbm{I}-A$不单进而不可逆，行列式$\det(\lambda \mathbbm{I}-A)=0$，此时称$\lambda$为$A$的特征值，数量为$n$。
\end{itemize}
\begin{definition}[正则算子]
设$X,Y$是赋范线性空间，定义算子$A \in \mathscr{B}(X,Y)$，若$R(A)=Y$且$A^{-1}\in \mathscr{B}(X,Y)$，则称$A$是正则算子。
\end{definition}
现在考虑无穷维时，设$X$是 Banach 空间，定义算子$A \in \mathscr{B}(X)$，对$\forall \lambda \in \mathbb{C}$考虑方程$(\lambda \mathbbm{I}-A)x=0$：
\begin{itemize}
\item 1. 如果该方程只有零解，那么$\lambda \mathbbm{I}-A$是单射，因此$\lambda \mathbbm{I}-A$“\href{https://math.fandom.com/zh/wiki/%E8%B0%B1}{可逆}”\footnote{很多地方都将可逆的扩大化了，严格意义上满射要求算子A的值域$R(A)=Y$，但如果仅在$R(A)$上那么可以宽松地说单射的算子可逆。}，分三种情况：
\begin{itemize}
\item 1. $R(\lambda \mathbbm{I}-A)=X$，因此$(\lambda \mathbbm{I}-A)^{-1} \in \mathscr{B}(X)$，记$\lambda\in\rho(A)$为正则值，$\rho(A)$称为预解集；
\item 2. $R(\lambda \mathbbm{I}-A) \neq X$但$\overline{R(\lambda \mathbbm{I}-A)}=X$，$(\lambda \mathbbm{I}-A)^{-1}$存在但不有界，记$\lambda\in\sigma_c(A)$为连续谱；
\item 3. $\overline{R(\lambda \mathbbm{I}-A)} \neq X$，$(\lambda \mathbbm{I}-A)^{-1}$存在但无法判断是否有界，记$\lambda\in\sigma_r(A)$为连续谱。
\end{itemize}
\item 2. 如果方程有非零解，那么$\lambda \mathbbm{I}-A$不是单射，因此$\lambda \mathbbm{I}-A$不可逆，记$\lambda\in\sigma_p(A)$为点谱（特征值）。
\end{itemize}
\begin{definition}[算子的谱]
定义算子$A$的谱为$\sigma(A)=\mathbb{C}-\rho(A)=\sigma_p(A)\cup\sigma_c(A)\cup\sigma_r(A)$。
\end{definition}

\begin{example}
\quad 在$X=L^2[0,1]$，上利用 \href{https://math.fandom.com/zh/wiki/Fourier_%E7%BA%A7%E6%95%B0}{Fourier 分解}定义闭算子[定义\ref{definition:bsz}] $A: \ X \to X$：
\[u(t) \quad \mapsto \quad -\dv[2]{u}{t}=-\dv[2]{t}\sum_{n=-\infty}^{\infty}e^{2i\pi nt}\int_0^1u(t)e^{2i\pi nt}\dd{t}=\sum_{n=-\infty}^{\infty}(2n\pi)^2e^{2i\pi nt}\int_0^1u(t)e^{2i\pi nt}\dd{t}\]
显然$\sigma(A)=\sigma_p(A)=\{(2n\pi)^2 \ | \ n \in \mathbb{Z}\}$，对应特征空间为$\{e^{2i\pi nt} \ | \ n\in\mathbb{Z}\}$。
\end{example}

\begin{example}\label{example:6.2}
\quad 在$X=C[0,1]$上定义算子 $A: \ X \to X \ , \ u(t) \mapsto tu(t)$，证明$\sigma(A)=\sigma_r(A)=[0,1]$。
\end{example}
\begin{proof}
对$\forall \lambda \in \mathbb{C}$都有$(\lambda\mathbbm{I}-A)u(t)=(\lambda-t)u(t)=0$可知方程只有零解$u(t)=0$，因此$(\lambda\mathbbm{I}-A)^{-1}$存在，若$\lambda \notin [0,1]$，显然有$(\lambda\mathbbm{I}-A)^{-1}u=u(t)/(\lambda-t)$，因此：
\[\|(\lambda\mathbbm{I}-A)^{-1}u\| \leq \sup_{t \in [0,1]}\frac{\|u\|}{\lambda-t}\]
进而可知$(\lambda\mathbbm{I}-A)^{-1} \in \mathscr{B}(X)$，即$\lambda$正则，可知$\sigma(A) \subset [0,1]$。若$\lambda \in [0,1]$，则$\overline{R(\lambda\mathbbm{I}-A)}=\overline{\{(\lambda-t)u(t) \ | \ u \in X\}}$，因为$\lambda-t \in X$，可知$\overline{R(\lambda\mathbbm{I}-A)}=\overline{\{u(t) \ | \ u \in X \ , \ u(\lambda)=0\}} \subset C[0,1]$，因此$\sigma(A)=\sigma_r(A)=[0,1]$。
\end{proof}

\begin{example}
\quad 在$X=L^2[0,1]$上类似上一个例子\ref{example:6.2}定义算子 $A: \ X \to X \ , \ u(t) \mapsto tu(t)$，则$\sigma(A)=\sigma_c(A)=[0,1]$。
\end{example}

\begin{theorem}\label{theorem:fskz}
设$X$是 Banach 空间，$A \in \mathscr{B}(X)$，则：
\begin{itemize}
\item 1. $\rho(A)$是开集，也即$\sigma(A)$是闭集；
\item 2. $\sigma(A) \subset \{\lambda \in \mathbb{C} \ | \ |\lambda| \leq \|A\|\}$或记为$r_{\sigma}(A) \leq \|A\|$。
\end{itemize}
\end{theorem}
\begin{proof}
由定理\ref{theorem:deltazz}对$\forall \lambda \in \rho(A)$，都$\exists \, \delta=\|\lambda\mathbbm{I}-A\|^{-1}$，使得对$\forall a<\delta$都有$(\lambda+a)\mathbbm{I}-A=(\lambda\mathbbm{I}-A)+a\mathbbm{I}$正则，即对$\forall \lambda \in \rho(A)$其开球$\{\lambda+a \ | \ |a|<\delta\} \subset \rho(A)$，因此$\rho(A)$是开集，第一部分得证。

对$\forall \lambda \in \mathbb{C}$满足$|\lambda|>\|A\|$，由定理\ref{theorem:nszyjxx}知$\lambda\mathbbm{I}-A=\lambda(\mathbbm{I}-A/\lambda)$正则，因此$\sigma(A) \subset \{\lambda \in \mathbb{C} \ | \ |\lambda| \leq \|A\|\}$。
\end{proof}

上述定理描述了算子的谱$\sigma(A)$实际上是个闭集且其谱半径$r_{\sigma}(A)$被算子范数$\|A\|$所控制，如上图所示。

\begin{definition}[谱半径]
谱半径$r_{\sigma}(A)$被定义为$r_{\sigma}(A)=\sup\{|\lambda| \ | \ \lambda \in \sigma(A)\}$。
\end{definition}

\begin{lemma}[谱映射定理]\label{lemma:pysdl}
设 $X$是 Banach 空间，$T\in B(X)$，$p$为多项式，则$\sigma(p(T))=p(\sigma(T))$。
\end{lemma}

\begin{theorem}[Gelfand 公式]\label{theorem:Gelfand}
设 $X$ 是 Banach 空间，$T\in\mathscr{B}(X)$ 是有界线性算子。谱半径$r_{\sigma}(T)$可以如下给出：
\[r_{\sigma}(T) = \lim_{n \to \infty} \|T^n\|^{1/n}\]
且等式右端的极限总是存在。
\end{theorem}
\begin{proof}
首先证明极限存在，令 $a_n = \ln \|T^n\|$。由算子范数的次可乘性 $\|T^{m+n}\| \le \|T^m\|\|T^n\|$ 取对数得$a_{m+n} \le a_m + a_n$，即$\{a_n\}$是次可加数列。对于次可加数列，有\href{https://math.fandom.com/zh/wiki/Fekete_%E5%BC%95%E7%90%86}{经典结论}：
\[\lim_{n \to \infty} \frac{a_n}{n} = \inf_{n \ge 1} \frac{a_n}{n}\]
由于$a_n/n=\ln\bigl(\|T^n\|^{1/n}\bigr)$ 且指数函数连续，故 $\lim\limits_{n \to \infty} \|T^n\|^{1/n}$ 存在，记该极限为 $r(T)$。对$\forall \lambda \in \sigma(T)$，由谱映射定理[引理\ref{lemma:pysdl}]知$\lambda^n \in \sigma(T^n)$。由定理\ref{theorem:fskz}知$|\lambda|^n \le \|T^n\|$。取 $n$ 次根得$|\lambda| \le \|T^n\|^{1/n}$。这对所有 $n$ 成立，故：
\[|\lambda| \le \inf_{n \ge 1} \|T^n\|^{1/n} = \lim_{n \to \infty} \|T^n\|^{1/n} = r(T)\]
对$\lambda \in \sigma(T)$ 取上确界得 $r_{\sigma}(T) \le r(T)$。

设$r > \rho(T)$，则$\{\lambda: |\lambda| = r\} \subset \rho(T)$，因此$(\lambda I - T)^{-1}$在$|\lambda| = r$上解析且有界。记：
\[M(r) = \max_{|\lambda| = r} \|(\lambda I - T)^{-1}\| < \infty\]
对 $|\lambda| > \|T\|$，定理\ref{theorem:nszyjxx}知$(\lambda I - T)^{-1}$有 \href{https://math.fandom.com/zh/wiki/%E8%B0%B1}{Neumann 级数展开}：
\[(\lambda I - T)^{-1} = \sum_{n=0}^{\infty} \frac{T^n}{\lambda^{n+1}}\]
该级数在算子范数下绝对收敛。由复分析的 Cauchy 积分公式，对$n \ge 0$，上述展开式的系数可由积分表示：
\[T^n = \frac{1}{2\pi i} \int_{|\lambda| = r} \lambda^n (\lambda I - T)^{-1} \, \dd\lambda \quad \Rightarrow \quad \|T^n\|\le \frac{1}{2\pi} \cdot \max_{|\lambda| = r} \|(\lambda I - T)^{-1}\| \cdot 2\pi r \cdot r^n = r^{n+1} M(r)\]
因此 $\|T^n\|^{1/n} \le r \cdot \bigl(r M(r)\bigr)^{1/n}$。令 $n \to \infty$，注意到 $\bigl(r M(r)\bigr)^{1/n} \to 1$，得$r(T) \le r$。由于$r > \rho(T)$是任意的，令$r \to \rho(T)^+$即得 $r(T) \le r_{\sigma}(T)$。综合下界与上界，得到 $r(T) = r_{\sigma}(T)$，即
\[r_{\sigma}(T) = \lim_{n \to \infty} \|T^n\|^{1/n}\]
\end{proof}

\begin{theorem}
设 $X$ 是 Banach 空间，$T\in\mathscr{B}(X)$是有界线性算子，则$\sigma(T) \neq \varnothing$。
\end{theorem}
\begin{proof}
假设$\sigma(T) = \varnothing$，对$\forall \lambda \in \mathbb{C}$，由于$\lambda \notin \sigma(T)$，算子 $\lambda I - T$ 可逆，因此$(\lambda I - T)^{-1} \in \mathscr{B}(X)$。对$\forall \lambda_0 \in \mathbb{C}$，当 $|\lambda - \lambda_0| < \|(\lambda_0 I - T)^{-1}\|^{-1}$ 时，利用 Neumann 级数展开有：
\[(\lambda I - T)^{-1}=\sum_{n=0}^{\infty} (\lambda_0 - \lambda)^n (\lambda_0 I - T)^{-n-1}\]
该级数在算子范数下收敛，故$(\lambda I - T)^{-1}$在$\lambda_0$处解析。由$\lambda_0$的任意性，$(\lambda I - T)^{-1}$在整个复平面解析，即为\href{https://math.fandom.com/zh/wiki/%E6%95%B4%E5%87%BD%E6%95%B0}{整函数}。当$|\lambda| > \|T\|$且$|\lambda| \to +\infty$时，
\[(\lambda I - T)^{-1} = \frac{1}{\lambda}\left(I - \frac{T}{\lambda}\right)^{-1}= \frac{1}{\lambda}\sum_{n=0}^{\infty}\left(\frac{T}{\lambda}\right)^n \quad \Rightarrow \quad \|(\lambda I - T)^{-1}\| \le \frac{1}{|\lambda|} \cdot \frac{1}{1 - \|T\|/|\lambda|}= \frac{1}{|\lambda| - \|T\|}\to0\]
因此$(\lambda I - T)^{-1}$在无穷远处趋于零，进而$(\lambda I - T)^{-1}$在$\mathbb{C}$上有界。由 \href{https://math.fandom.com/zh/wiki/Liouville_%E5%AE%9A%E7%90%86}{Liouville 定理}，有界整函数必为常值函数。故存在常数 $C \in \mathscr{B}(X)$ 使得 $(\lambda I - T)^{-1} \equiv C$。但前面已证$\|(\lambda I - T)^{-1}\|\to0$，从而$C=0$，即$(\lambda I - T)^{-1}\equiv 0$。然而零算子不可逆，因此假设不成立，$\sigma(T)$非空。
\end{proof}

\begin{example}
\quad 设$X=l^2$，定义算子$A \in \mathscr{B}(X): \ x=(\xi_1,\xi_2,\cdots) \mapsto Ax=(0,\xi_1,\xi_2,\cdots)$，显然$\|A\|=1$，有$\sigma_p(A)=\varnothing \ , \ \sigma_c(A)=\{\lambda\in\mathbb{C} \ | \ \|\lambda\|=1\} \ , \ \sigma_r(A)=\{\lambda\in\mathbb{C} \ | \ \|\lambda\|<1\}$。
\end{example}
\begin{proof}
首先由$\|A\|=1$可知$\sigma(A) \subset \{\lambda \ | \ |\lambda|\leq1\}$，其次考虑方程$(\lambda\mathbbm{I}-A)x=(\lambda\xi_1,\lambda\xi_2-\xi_1,\cdots)=0$，当$\lambda\neq0$时，显然$x=0$；当$\lambda=0$时，显然$x=0$。因此$\sigma_p(A)=\varnothing$。

注意到$A$的共轭算子$A^*: \ X \to X \ , \ x=(\xi_1,\xi_2,\cdots) \mapsto A^*x=(\xi_2,\xi_3,\cdots)$，考虑算子$(\lambda\mathbbm{I}-A)=\lambda^*\mathbbm{I}-A^*$，由共轭算子的定义可知对$\forall x \in X$都$\exists \, y \in X$由关系式$((\lambda\mathbbm{I}-A)x,y)=(x,(\lambda^*\mathbbm{I}-A^*)y)$一一对应。对$\forall y \in \overline{R(\lambda\mathbbm{I}-A)}^{\perp}=R(\lambda\mathbbm{I}-A)^{\perp}=\ker(\lambda^*\mathbbm{I}-A^*)$（定理\ref{theorem:zgeker}和性质\ref{itemize:zjbb}第三点），都有$0=(\lambda^*\mathbbm{I}-A^*)y=(\lambda^*\xi_1-\xi_2,\lambda^*\xi_2-\xi_3,\cdots)$，因此$y=\xi_1(1,\lambda^*,(\lambda^*)^2,\cdots)$。只有当$|\lambda|<1$时$y \in l^2$。

当$|\lambda|<1$时，$R(\lambda\mathbbm{I}-A)^{\perp}=\ker(\lambda^*\mathbbm{I}-A^*)\neq\{0\}$，通过一一映射后按照谱的定义可知$\{|\lambda|<1\} \subset \sigma_r(A)\subset\sigma(A)$。由于$\sigma(A)$是闭集[定理\ref{theorem:fskz}]，因此$\sigma(A) \supset \overline{\{|\lambda|<1\}}=\{|\lambda|\leq1\}$，结合最开头的结论可知$\sigma(A)=\{|\lambda|\leq1\}$。

当$|\lambda|=1$时，$\ker(\lambda^*\mathbbm{I}-A^*)=R(\lambda\mathbbm{I}-A)^{\perp}=\overline{R(\lambda\mathbbm{I}-A)}^{\perp}=\{0\}$，因此$\overline{R(\lambda\mathbbm{I}-A)}=X$，由$\sigma(A)=\{|\lambda|\leq1\}$可知$R(\lambda\mathbbm{I}-A)\neq X$，因此$\sigma_c(A)=\{|\lambda|=1\}$，进而可知$\sigma_r(A)=\{|\lambda|<1\}$。

或者可以直接证明当$|\lambda|=1$时$R(\lambda\mathbbm{I}-A)$不是闭的，取点列$\{x_n\}$：
\[x_n=\frac{1}{\sqrt{n}}\left(\lambda^*,(\lambda^*)^2,\cdots,(\lambda^*)^n,0,\cdots\right)\]
计算：
\[(\lambda\mathbbm{I}-A)x_n=\left(\frac{1}{\sqrt{n}},\frac{\lambda^*}{\sqrt{n}}-\frac{\lambda^*}{\sqrt{n}},\cdots,\frac{(\lambda^*)^{n-1}}{\sqrt{n}}-\frac{(\lambda^*)^{n-1}}{\sqrt{n}},-\frac{(\lambda^*)^n}{\sqrt{n}},0,\cdots\right)=\left(\frac{1}{\sqrt{n}},0,\cdots,0,-\frac{(\lambda^*)^n}{\sqrt{n}},0,\cdots\right)\]
因此其范数$\|(\lambda\mathbbm{I}-A)x_n\|=\sqrt{2/n} \to 0$，说明$\lambda\mathbbm{I}-A$没有正的下界，因此其值域不闭。
\end{proof}

\section{Riesz-Fredholm 定理}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，考虑算子$T=\mathbbm{I}-A$，先考察有限维的情况，如取$X=\mathbb{R}^n$，考虑方程$Tx=y$。那么有以下两种状况，由 Fredholm 二择一定理所描述。
\begin{theorem}[Fredholm 二择一]
设$X=\mathbb{R}^n$，$x,y \in X$，$T$是$X$上的算子，考虑线性方程组$Tx=y$，以下两种情况恰居其一：
\begin{enumerate}
\item 如果$\ker T = \{\mathbf{0}\}$，则对任意$y$方程存在唯一解；
\item 如果$\ker T \neq \{\mathbf{0}\}$，则方程可解的充要条件是$y \perp \ker T^*$。
\end{enumerate}
此时必有指标$\operatorname{ind}(A) = \dim\ker T-\dim\ker T^*=0$。
\end{theorem}
第一点显然[定理\ref{theorem:djm}]，考虑第二点，也即考虑$T$不是单射，我们令$\{y_1,y_2,\cdots,y_p\}$是$R(T)$的一组基，可以扩充为$\{y_1,y_2,\cdots,y_p,z_1,z_2,\cdots,z_q\}$为$X$的一组基，也即$X=R(T)\oplus X_1$，其中$X_1=\text{span}\{z_1,z_2,\cdots,z_q\}$。

定义对偶空间$X^*=R(T)^*\oplus X_1^*$，设$\{y_1^*,y_2^*,\cdots,y_p^*,z_1^*,z_2^*,\cdots,z_q^*\}$为$X^*$的一组基对偶基，即满足$y_i^*(y_j)=z_i^*(z_j)=\delta_{ij} \ , \ y_i^*(z_j)=z_i^*(y_j)=0$，其中$X_1^*=\text{span}\{z_1^*,z_2^*,\cdots,z_q^*\}$。因此$y \in R(T)$等价于对$\forall f \in X_1^*$都有$f(y)=y(f)=0$，即$X_1^*=\ker T^*$。因此原方程有解等价于$y \in (\ker T^*)^{\perp}$。
\begin{definition}
\begin{itemize}
\item 1. $\forall M \subset X$，记$^{\perp}M=\{f \in X^* \ | \ f(x)=0 \ , \ \forall x \in M\}$；
\item 2. $\forall N \subset X^*$，记$N^{\perp}=\{x \in X \ | \ f(x)=0 \ , \ \forall f \in N\}$。
\end{itemize}
\end{definition}
\begin{lemma}\label{lemma:crazy}
对$\forall T \in \mathscr{B}(X)$，都有$\overline{R(T)}=(\ker T^*)^{\perp}$和$\overline{R(T^*)}=^{\perp}\ker T$。
\end{lemma}
\begin{proof}
对$\forall T \in \mathscr{B}(X,Y=X)$，按照定义分别写出$\ker T$，$\ker T^*$，$^{\perp}R(T)$，$R(T^*)^{\perp}$：
\[\begin{aligned}
\ker T         &= \{x \in X \ | \ Tx=0\}=\{x \in X \ | \ g(Tx)=(T^*g)x=0 \ , \ \forall g \in Y^*\}=\{x \in X \ | \ f(x)=0 \ , \ \forall f \in X^*\}\\
\ker T^*       &= \{f \in Y^* \ | \ T^*f=0\}=\{g \in X^* \ | \ f(x)=0 \ , \ \forall x \in X\}\\
^{\perp}R(T)   &= \{f \in X^* \ | \ f(x)=0 \ , \ \forall x \in R(T)\}\\
R(T^*)^{\perp} &= \{x \in X \ | \ f(x)=0 \ , \ \forall f \in R(T^*)\}\\
\end{aligned}\]
显然有$^{\perp}R(T) \subseteq \ker T^*$，$R(T^*)^{\perp} \subseteq \ker T$成立。

对$\overline{R(T)} \subseteq (\ker T^*)^{\perp}$由正交补的闭性[性质\ref{itemize:zjbb}第三点]只需证$R(T) \subseteq (\ker T^*)^{\perp}$。对$\forall Tx \in R(T) \subset Y=X \ , \ f \in \ker T^*$都有$f(Tx)=(T^*f)x=0$，即$Tx \in (\ker T^*)^{\perp}$，因此$R(T) \subseteq (\ker T^*)^{\perp}$，进而$\overline{R(T)} \subseteq (\ker T^*)^{\perp}$。

类似的对$\overline{R(T^*)} \subseteq ^{\perp}\ker T$只需证$R(T^*) \subseteq ^{\perp}\ker T$。对$\forall f \in R(T^*) \ , \ x \in \ker T$都有$f(x)=0$，即$f \in ^{\perp}\ker T$，因此$R(T^*) \subseteq ^{\perp}\ker T$，进而$\overline{R(T^*)} \subseteq ^{\perp}\ker T$。

利用正交补的传递性[性质\ref{itemize:zjbb}第四点]可得：
\[\overline{R(T)} \subseteq (\ker T^*)^{\perp} \subseteq (^{\perp}R(T))^{\perp} \quad , \quad \overline{R(T^*)} \subseteq ^{\perp}\ker T \subseteq ^{\perp}(R(T^*)^{\perp})\]

由 Hahn-Banach 定理推论[命题\ref{proposition:hb3}]可知若取$M=R(T)$，若$x_0 \in X-R(T)$到$R(T)$的距离$d$都为正，则存在$f \in X^*$使得$f(x_0)=d$且$f(R(T))=0$。由$f \in X^*$的连续性这里可以扩展到闭包，上述命题等价于若$x_0 \in \overline{R(T)}$，则对$\forall f \in X^*$使得$f(R(T))=0$都有$f(x_0)=0$。对$\forall y_0 \in (^{\perp}R(T))^{\perp}$，其满足对$\forall f \in ^{\perp}R(T)$都有$f(y_0)=0$，而$^{\perp}R(T)=\{f \in X^* \ | \ f(x)=0 \ , \ \forall x \in R(T)\}$，即$y_0 \in \overline{R(T)}$。因此有$(^{\perp}R(T))^{\perp} \subseteq \overline{R(T)}$。综上所述，$(^{\perp}R(T))^{\perp}=\overline{R(T)}$，夹逼可知$\overline{R(T)}=(\ker T^*)^{\perp}$。

类似的，对$\forall f_0 \in ^{\perp}(R(T^*)^{\perp})$，其满足对$\forall x \in R(T^*)^{\perp}$都有$f_0(x)=0$，而$R(T^*)^{\perp}=\{x \in X \ | \ f(x)=0 \ , \ \forall f \in R(T^*)\}$。即$f_0 \in R(T^*) \subset \overline{R(T^*)}$。因此有$^{\perp}(R(T^*)^{\perp}) \subseteq \overline{R(T^*)}$，夹逼可知$\overline{R(T^*)}=^{\perp}(R(T^*)^{\perp})$。
\end{proof}
\begin{remark}
\quad 对任意子空间$M \subset X$都有$(^{\perp}M)^{\perp}=\overline{M}$；若闭子空间$M \subset X \ , \ N \subset X^*$，则$^{\perp}M=N \ \Leftrightarrow \ M=N^{\perp}$。
\end{remark}


\begin{theorem}[Riesz-Fredholm 定理]\label{theorem:Riesz-Fredholm}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，考虑算子$T=\mathbbm{I}-A$和$T^*=\mathbbm{I}-A^*$，则：
\begin{itemize}
\item 1. $\sigma(T)=\sigma(T^*)$；
\item 2. $R(T)=(\ker T^*)^{\perp}$，$R(T^*)=^{\perp}(\ker T)$；
\item 3. 若$\ker T=\{0\}$，则$R(T)=X$；
\item 4. $\dim\ker T=\dim\ker T^*<+\infty$。
\end{itemize}
\end{theorem}
\begin{remark}
\quad Riesz-Fredholm 定理告诉我们紧算子$A$使得$T$的行为与有限维算子很相似。下面我们分步证明：
\end{remark}
\begin{proposition}\label{proposition:pdj1}
1. $\sigma(T)=\sigma(T^*)$。
\end{proposition}
\begin{proof}
要证明$\sigma(T) = \sigma(T^*)$，等价于证明$\rho(T) = \rho(T^*)$，也即对$\forall \lambda \in \rho(T)$证明$(\lambda\mathbbm{I}-T)^{-1}\in\mathscr{B}(X) \ \Leftrightarrow \ (\lambda\mathbbm{I}-T^*)^{-1}\in\mathscr{B}(X)$，进而等价于证明$T^{-1}\in\mathscr{B}(X) \ \Leftrightarrow \ (T^*)^{-1}\in\mathscr{B}(X)$。当然还要证$R(T)=X$。

对必要性，$(T^*)^{-1}=(T^{-1})^*\in\mathscr{B}(X)$；对充分性，由必要性可知$(T^*)^{-1}\in\mathscr{B}(X) \ \Rightarrow \ (T^{**})^{-1}\in\mathscr{B}(X)$，再由典型映射是单的，即$T=T^{**}|_X$，在$R(T)$上可定义$T^{-1}$，即证。

对$\forall x \in X$，取柯西列$\{x_n\} \subset X$满足$x_n \to x \in X$，记$\{y_n\}=\{Tx_n\}$。由$T\in\mathscr{B}(X)$知对$\forall n,m \in \mathbb{N}_+$：
\[\|y_n-y_m\|=\|Tx_n-Tx_m\|\leq\|T\|\cdot\|x_n-x_m\| \to 0\]
因此$\{y_n\} \subset R(T)$是柯西列，记$y_n \to y$，因为$T\in\mathscr{B}(X)$显然有$Tx_n \to Tx \in R(T)$，考虑：
\[\|Tx_n-y\| \leq \|Tx_n-y_n\|+\|y_n-y\|=\|y_n-y\| \to 0 \quad \Rightarrow \quad y=Tx \in R(T)\]
因此$R(T)$中的任意柯西列$\{y_n\}$都收敛到其自身，故$R(T)$是闭的。使用反证法证明$R(T)=X$，则$\exists \, x_0 \in X-R(T)$，由 Hahn-Banach 定理推论[命题\ref{proposition:hb3}]，$\exists \, f \in X^*$满足$f(x_0)=\|x_0\|\neq 0$且对$\forall y \in R(T)$都有$f(y)=0$，即对$\forall x \in X \ , \ y=Tx \in R(T)$都有$f(y)=f(Tx)=(T^*f)x=0$，因此$T^*f=0$即$f \in \ker T^*$，但是上面证明了$T^*$可逆，因此$\ker T^*=\{0\}$，因此$f=0$，得出矛盾，故$R(T)=X$。
\end{proof}

\begin{definition}[闭值域算子]
设$X$为赋范线性空间，$T \in \mathscr{B}(X)$，若$R(T)$是$X$的闭子空间，则称$T$是闭值域算子。
\end{definition}
\begin{theorem}\label{theorem:jszb}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，则算子$T=\mathbbm{I}-A$是闭值域算子。
\end{theorem}
\begin{proof}
由于 $T = \mathbb{I} - A$ 连续，核 $\ker T$ 是 $X$ 的闭子空间。考虑$\widetilde{T}: \ \widetilde{X}=X/\ker T \to X \ , \ [x] \mapsto Tx$，商空间$\widetilde{X}$ [定义\ref{definition:skj}]的元素为代表元$[x]=x+\ker T=\{x+z \ | \ z \in \ker T\}$，其范数定义为：
\[\|[x]\|=\inf_{y \in [x]}\|y\|=\inf_{z \in \ker T}\|x+z\|=d(x,\ker T)\]
显然$\ker\widetilde{T}=\{[0]\}$，$\widetilde{T}$有界线性且为单射。同时$R(\widetilde{T})=R(T)$，在值域$R(\widetilde{T})$可以定义逆$\widetilde{T}^{-1}$。要证 $R(T)$ 闭，只需证$\widetilde{T}^{-1}$有界，即$\exists \, c > 0$对$\forall x \in X$都有$\|[x]\| \le c \|Tx\|$。因为若$\widetilde{T}^{-1}$有界，则$R(T)$同构于$\widetilde{X}$，是 Banach 空间[定理\ref{theorem:skjwb}]，因此$R(T)$也是 Banach 空间，进而是闭的。采用反证法，假设$\widetilde{T}^{-1}$无界，则对$\forall n \in \mathbb{N}$都$\exists \, [x_n] \in \widetilde{X}$使得$\|[x_n]\| \ge n \|Tx_n\|$，可以取标准化$\|[x]\|=1$，则$\|\widetilde{T}[x_n]\|=\|Tx_n\| \to 0$。

取$[x_n]$中的代表元为$x_n$满足$\|x_n\| \leq 2$，由$A$紧可知$\{Ax_n\}$有收敛子列$Ax_{n_k} \to z$。另一方面$\widetilde{T}x_{n_k}=Tx_{n_k}=x_{n_k}-Ax_{n_k} \to 0$，可知$x_{n_k} \to z$，则$Az=z$即$Tz=z-Az=0$，因此$z \in \ker T$，即$\|[x]\|=\|[0]\|=0$。
但$\|[x]\|=1$，进而得到矛盾，因此$\widetilde{T}^{-1}$有界，原命题得证。
\end{proof}

\begin{proposition}\label{proposition:6-2}
2. $R(T)=(\ker T^*)^{\perp}$，$R(T^*)=^{\perp}(\ker T)$。
\end{proposition}
\begin{proof}
由引理\ref{lemma:crazy}可知，证明第二点只需证明$T$和$T^*$是闭值域算子。由定理\ref{theorem:jszb}即证前一半。由 Schauder 定理[定理\ref{theorem:Schauder}]知$A^* \in \mathscr{C}(X^*)$，再由定理\ref{theorem:wb}知$X^*$是 Banach 空间，因此$T^*$也是闭值域算子，后一半也即证。
\end{proof}

\begin{proposition}\label{proposition:djm3}
3. 若$\ker T=\{0\}$，则$R(T)=X$。
\end{proposition}
\begin{proof}
由定理\ref{theorem:jszb}可知$R(T)$是$X$的闭子空间，且$\exists \, c>0$使得$\|Tx\| \geq \alpha\|x\|$。反证法，设$R(T) \subset X$，令$X_0=X \ , \ X_k=TX_{k-1} \ , \ k \in \mathbb{N}_+$，则$\cdots \subset X_k \subset X_{k-1} \subset \cdots \subset X_1 \subset X_0$，这里都是真包含关系因此$X_k$是$X_{k-1}$的真闭子空间，则由 Riesz 引理[定理\ref{Riesz}]，$\exists \, x_k \in X_{k-1}-X_k$满足$\|x_k\|=1$，都有$d(x_k,X_k)>1/2$。由$A \in \mathscr{C}(X)$可知$\{Ax_k\}$有收敛子列仍记为$\{Ax_k\}$，但是$\|Ax_k-Ax_{k+p}\|=\|x_k-(Tx_k+x_{k+p}-Tx_{k+p})\| \geq d(x_k,X_k)>1/2$，其中$Tx_k \in X_k \ , \ x_{k+p} \in X_{k+p-1} \subset X_k \ , \ Tx_{k+p} \in X_{k+p} \subset X_k \ \Rightarrow \ Tx_k+x_{k+p}-Tx_{k+p} \in X_k$，矛盾，命题得证。
\end{proof}

\begin{proposition}
4. $\dim\ker T=\dim\ker T^*<+\infty$。
\end{proposition}
\begin{proof}
先考虑单射即命题\ref{proposition:djm3}成立，由逆算子定理[定理\ref{theorem:nszdl}]可知$T^{-1} \in \mathscr{B}(X)$，因此$0 \in \rho(T)$，由命题\ref{proposition:pdj1}知$0 \in \rho(T^*)$，因此$(T^*)^{-1} \in \mathscr{B}(X)$，从而$\ker T^*=\{0\}$，因此$\dim\ker T=\dim\ker T^*=0$。

对非单射情况分步证明：
\begin{lemma}[$\dim\ker T,\dim\ker T^*<+\infty$]\label{lemma:4.1}
令$X_0=\ker T$，$B_0=\{x \in X_0 \ | \ \|x\|<1\}$是$X_0$的单位球，对$\forall x \in B_0$都有$0=Tx=x-Ax$，即$x=Ax \in A(B_0)$，因此$B_0 \subset A(B_0)$列紧（列紧集的子集也列紧），从而$X_0$有限维。$T^*$证明同理。
\end{lemma}

\begin{lemma}[若有限维子空间$X_0 \subset X$则存在闭子空间$X_1 \subset X$使得$X=X_0 \oplus X_1$]\label{lemma:4.2}
设$\{x_i\}_{i=1}^n$是$X_0$的一组基，则由 Hahn-Banach 定理[定理\ref{HB}]，存在$\{f_i\}_{i=1}^n \subset X^*$使得$f_i(x_j)=\delta_{ij}$。令：
\[X_1=\bigcap_{i=1}^n\ker f_i\]
则$X_1$闭，且$X_0 \cap X_1=\{0\}$（包证分解唯一）。对$\forall x \in X$都$\exists \, ! \, x_0 \in X_1$使得：
\[x=x_0+\sum_{i=1}^nf_i(x)x_i\]
\end{lemma}

\begin{lemma}[设$\{f_i\}_{i=1}^m \subset X^*$是线性无关集，则$\exists \, \{y_i\}_{i=1}^m \subset X$使得$f_i(y_j)=\delta_{ij}$]\label{lemma:4.3}
考虑映射$F: \ X \to \mathbb{K}^m \ , \ x \mapsto F(x)=(f_1(x),\cdots,f_m(x))$，只需证明$F$为满射即证所求。采用反证法，若$F(X) \subset \mathbb{K}^n$，则由 Hahn-Banach 定理[定理\ref{HB}]存在$0 \neq \alpha=(\alpha_1,\cdots,\alpha_m)\in\mathbb{K}^*=\mathbb{K}$使得对$\forall x \in X$都有：
\[\alpha|_{F(X)}=(\alpha,F(x))=\sum_{i=1}^m\alpha_if_i(x)=0\]
即$\{f_i\}_{i=1}^m$线性相关，得到矛盾，故原命题得证。
\end{lemma}

根据引理\ref{lemma:4.1}设$\dim\ker T=n \ , \ \dim\ker T^*=m$。取$\ker T$的一组基$\{x_i\}_{i=1}^n$和$\ker T^*$的一组基$\{f_i\}_{i=1}^m$。

设$n<m$，存在闭子空间$X_1$使得$X=\ker T \oplus X_1$[引理\ref{lemma:4.2}]，取$\{y_i\}_{i=1}^m \subset X$使得$f_i(y_j)=\delta_{ij}$[引理\ref{lemma:4.3}]。令：
$$\widetilde{T}: \ X=\ker T \oplus X_1 \to \text{span}\{y_i\}_{i=1}^n \oplus R(T), \quad x=x_0+\sum_{i=1}^n\phi_i(x)x_i \quad \mapsto \quad \widetilde{T}x=Tx_0+\sum_{i=1}^n\phi_i(x)y_i$$
显然$\ker\widetilde{T}=\{0\}$，且$\widetilde{A}x=x-\widetilde{T}x=Ax_0+\sum_{i=1}^n\phi_i(x)(x_i-y_i)$为紧算子（前一半显然，后一半有穷秩）[定理\ref{theorem:yqzj}]。此时$R(\widetilde{T})=X$[命题\ref{proposition:djm3}]但$\{y_{n+1},\cdots,y_m\} \notin R(\widetilde{T})$，矛盾，故$\dim\ker T^* \leq \dim\ker T$。同理$\dim\ker T^{**} \leq \dim\ker T^*$。由$X \subset X^{**}$得$\ker T \subset \ker T^{**}$，进而$\dim\ker T \leq \dim\ker T^{**} \leq \dim\ker T^*$，综上所述原命题得证。
\end{proof}
\begin{remark}
\quad 由命题\ref{proposition:6-2}可知$R(T^*)=(\ker T^{**})^{\perp}=^{\perp}(\ker T)$，因此$\dim\ker T=\dim\ker T^{**}$。
\end{remark}

\section{紧算子谱}\label{jszp}
\begin{theorem}
设$X$是无限维 Banach 空间，$A \in \mathscr{C}(X)$，则：
\begin{itemize}
\item 1. $0 \in \sigma(A)$；
\item 2. $\sigma(A)=\{0\}\cup\sigma_p(A)$；
\item 3. $\sigma(A)$以0作为聚点，即对$\forall \delta>0$，$\{\lambda\in\sigma(A) \ | \ |\lambda|\geq\delta\}$为有限集。
\end{itemize}
\end{theorem}
\begin{remark}
\quad 第三点可以等价于对$\forall \delta>0$，取指标集$\Lambda=\{\lambda\in\sigma_p(A) \ | \ |\lambda|>\delta\}$，则$\bigcup\limits_{\lambda\in\Lambda}\ker(\lambda\mathbbm{I}-A)$是有限集。
\end{remark}
\begin{proof}
第一点等价于$0 \notin \rho(A)$，如若不然，则$\mathbbm{I}=(0\cdot\mathbbm{I}-A)(0\cdot\mathbbm{I}-A)^{-1}=AA^{-1}\in\mathscr{C}(X)$[定理\ref{theorem:szfh}]，矛盾。

第二点，取$\forall \lambda \in \sigma(A)-\{0\}$，要么$\ker(\lambda\mathbbm{I}-A)\neq\{0\}$，则$\lambda\in\sigma_p(A)$，要么$\ker(\lambda\mathbbm{I}-A)=\{0\}$，由 Riesz-Fredholm 定理[定理\ref{theorem:Riesz-Fredholm}]，$R(\lambda\mathbbm{I}-A)=X$，再由逆算子定理[定理\ref{theorem:nszdl}]可知$\lambda\in\rho(A)$。

第三点，采用反证法，设存在可数无限集$\{\lambda_n\}$使得$\lambda_n>\delta$，取$\{x_n \in \ker(\lambda_n\mathbbm{I}-A)-\{0\}\}$满足$Ax_n=\lambda_nx_n$。采用归纳法证明$\{x_n\}$线性无关，假设已知$\{x_1,\cdots,x_n\}$线性无关，取：
\[y=\sum_{i=1}^{n+1}a_ix_i=0 \quad \Rightarrow \quad Ay=\sum_{i=1}^{n+1}a_i\lambda_ix_i \quad \Rightarrow \quad \lambda_{n+1}y-Ay=\sum_{i=1}^{n+1}(\lambda_{n+1}-\lambda_i)a_ix_i=0\]
因此对$i=1,\cdots,n+1$都有$a_i=0$，即$\{x_1,\cdots,x_{n+1}\}$线性无关，由数学归纳法可证。令$V_n=\text{span}\{x_1,\cdots,x_n\}$，$V_n$是$V_{n+1}$的真子空间，由 Riesz 引理[定理\ref{Riesz}]，$\exists \, \{y_n\}$满足$y_n \in V_{n+1}-V_n$且$\|y_n\|=1$使得$d(y_n,V_n)>1/2$。由$A \in \mathscr{C}(X)$，则$\{Ay_n\}$列紧，但对$\forall n,p \in \mathbb{N}_+$，记$y_{n+p}=\alpha x_{n+p+1}+u$，其中$u \in V_{n+p}$：
\[\|Ay_{n+p}-Ay_n\|=\|\alpha\lambda_{n+p+1}x_{n+p+1}+A(u-y_n)\|>\alpha\lambda_{n+p+1}d(x_{n+p+1},V_{n+p})>\frac{\alpha\lambda_{n+p+1}}{2}>0\]
这里$y_n \in V_{n+1} \subset V_{n+p}$，因此$u-y_n \in V_{n+p}$，由于$V_{n+p}$是$A$的特征向量张成的空间，因此$AV_{n+p}=V_{n+p}$，即$V_{n+p}$是$A$不变的，故$A(u-y_n) \in V_{n+p}$，这与$\{Ay_n\}$列紧矛盾，因此假设不成立，原命题得证。
\end{proof}
\begin{remark}
\quad 对第三点，记$S_n=\{\lambda\in\sigma_p(A) \ | \ |\lambda|>1/n\}$，则$\sigma_p(A)=\bigcup\limits_{n=1}^{+\infty}S_n$，因为$S_n$有限，可知$\sigma_p(A)$至多可数。
\end{remark}
由本定理和 Riesz-Fredholm 定理[定理\ref{theorem:Riesz-Fredholm}]可知，紧算子只有非零的点谱，这与有限维算子可以说几乎一样了，我们可以通过类似 Jordan 分解的办法研究紧算子的结构，首先从有限维开始叙述。先引入不变子空间：
\begin{definition}
设$X$是 Banach 空间，$T \in \mathscr{B}(X)$，若子空间$M \subset X$满足$TM \subseteq M$则称$M$是$T$的不变子空间。
\end{definition}
\begin{itemize}
\item 1. $\{0\}$和$X$是$T$的平凡子空间；
\item 2. 若$M$是不变子空间，则$\overline{M}$是闭的不变子空间（利用$T$的连续性）；
\item 3. 若$\lambda \in \sigma_p(T)$则$\ker(\lambda\mathbbm{I}-T)$是$T$的不变子空间（下一点的特殊情况）；
\item 4. 若$S \in \mathscr{B}(X)$与$T$可交换，即$AB=BA$，则$\ker B$和$R(B)$是$A$的不变子空间。
\end{itemize}
\begin{proof}
第四点：对$x \in \ker B$，$ABx=BAx=0$，即$Ax \in \ker B$；对$y=Bx \in B$，$Ay=A(Bx)=B(Ax)\in R(B)$。
\end{proof}

\begin{theorem}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，若$\dim X \geq 2$则$A$必有非平凡不变子空间。
\end{theorem}

\begin{example}
\quad $A: \ \mathbb{R}^2 \to \mathbb{R}^2$，矩阵$\begin{pmatrix}1 & -1 \\ 0 & 1 \end{pmatrix}$不可对角化，计算方程$(\lambda\mathbbm{I}-A)x=0$可以求解特征值为$\lambda=1$，特征向量为$\{0,\mathbf{e}_x=\begin{pmatrix}1\\0\end{pmatrix}\}$，因此$\ker(\mathbbm{I}-A)=R(\mathbbm{I}-A)=\text{span}\{\mathbf{e}_x\}$，即$\dim\ker(\mathbbm{I}-A)=\operatorname{codim} R(\mathbbm{I}-A)=n-\dim R(\mathbbm{I}-A)$，但是$X \neq \ker(\mathbbm{I}-A) \oplus R(\mathbbm{I}-A)$而$X=\ker(\mathbbm{I}-A)^2 \oplus R(\mathbbm{I}-A)^2$。
\end{example}

\begin{lemma}\label{lemma:mlfj}
设$T$是有限维空间$M^m$上的$p$次幂零变换，即$T^p=0$但对$i<p$都有$T^i \neq 0$，则$\exists \, 1 \leq p_1 \leq p_2 \leq \cdots \leq p_k=p$以及基底$\{x_1,Tx_1,\cdots,T^{p_1-1}x_1,\cdots,x_k,Tx_k,\cdots,T^{p_k-1}x_k\}$使得$T \cong \operatorname{diag}\{B_1,B_2,\cdots,B_k\}$，其中：
\[B_i=\begin{pmatrix}
0 & 1 & \cdots & 0 \\
0 & 0 & \ddots & \vdots \\
\vdots & \ddots & \ddots & 1 \\
0 & \cdots & 0 & 0
\end{pmatrix}_{p_i \times p_i}, \quad \sum_{i=1}^kp_k=m\]
\end{lemma}
\begin{proof}
由$\forall i<p_k$都有$T^i \neq 0$可知，$\exists \, x_k \in M$使得$\{x_k,Tx_k,\cdots,T^{p_k-1}x_k\}$线性无关，因为若：
\[0=\sum_{j=1}^{p_k}a_jT^{j-1}x_k \quad \Rightarrow \quad T^{p_k-1}0=T^{p_k-1}\sum_{j=1}^{p_k}a_jT^{j-1}x_k=a_1T^{p_k-1}x_k=0 \quad \Rightarrow \quad a_1=0\]
递推可知$a_1=a_2=\cdots=a_{p_k}=0$。记$E_k=\text{span}\{x_k,Tx_k,\cdots,T^{p_k-1}x_k\}$，$E_k$是$T$的闭不变子空间，若$E_k=M$则命题得证。若$E_k \subset M$则令$M_1=M/E_k$，定义其维数为$m-p_k$。诱导变换$\overline{T}: \ \overline{M}\to\overline{M} \ , \ [x] \mapsto \overline{T}[x]=Tv+E_k$。易见$\overline{T}$是幂零的，且其幂零指数不超过$p_k$，记为$p_{k-1}$。同上可得，可以构造出一个新的$T$不变子空间记为$E_{k-1}=\text{span}\{[x_{k-1}],\overline{T}[x_{k-1}],\cdots,\overline{T}^{p_{k-1}-1}[x_{k-1}]\}$，取代表元为$x_{k-1}$，则$E_{k-1}=\text{span}\{x_{k-1},Tx_{k-1},\cdots,T^{p_{k-1}-1}x_{k-1}\}$，经过有限次重复后可得$\exists \, 1 \leq p_1 \leq p_2 \leq \cdots \leq p_k=p$以及基底$\{x_1,Tx_1,\cdots,T^{p_1-1}x_1,\cdots,x_k,Tx_k,\cdots,T^{p_k-1}x_k\}$使得：
\[M=\bigoplus_{i=1}^kE_i=\bigoplus_{i=1}^k \operatorname{span}\{x_i,Tx_i,\cdots,T^{p_i-1}x_i\}\]
在该基底下$T \cong \operatorname{diag}\{B_1,B_2,\cdots,B_k\}$，$M$的完全分解由商空间的完备性[定理\ref{theorem:skjwb}]保证。
\end{proof}

\begin{lemma}\label{lemma:mlsz}
设$X=\mathbb{R}^n$，$T \in \mathscr{B}(X)$，则$\exists \, 1 \leq p \leq n$使得$\ker T^p=\ker T^{p+1}=\cdots$
\end{lemma}
\begin{proof}
由$R(T) \subseteq X$可知$\{0\} \subseteq \ker T \subseteq \ker T^2 \subseteq \cdots$，由$\dim\ker T^n \leq \dim X<+\infty$，存在一个最小的$p \leq n$使得$\ker T^p=\ker T^{p+1} \subseteq \ker T^{p+2}$，对$\forall x \in \ker T^{p+2}$都有$T^{p+2}x=T^{p+1}(Tx)=0$，即$Tx \in \ker T^{p+1}=\ker T^p$，因此$x \in \ker T^{p+1}$，即$\ker T^{p+2} \subseteq \ker T^{p+1}$，原命题得证。
\end{proof}
\begin{remark}
\quad 上述求得的$p$显然满足$X=\ker T^p \oplus R(T^p)$，且$T|_{R(T^p)}$存在有界逆。本引理可以用来构造幂零矩阵。
\end{remark}
\begin{theorem}[Cayley–Hamilton 定理]\label{theorem:Cayley–Hamilton}
设$M$是$n \times n$矩阵，其特征多项式定义为$p(\lambda)=\det(\lambda\mathbbm{I}_n-M)$，则$p(M)=\mathbf{0}$。
\end{theorem}

\begin{theorem}[有限维 Jordan 分解定理]
设$X=\mathbb{R}^n$，$A \in \mathscr{B}(X)$，存在基使得$A$对应矩阵$A \cong \operatorname{diag}\{A_1,A_2,\cdots,A_k\}$，其中$A_i$为 Jordan 块：
\[A_i=\begin{pmatrix}
\lambda_i & 1 & \cdots & 0 \\
0 & \lambda_i & \ddots & \vdots \\
\vdots & \ddots & \ddots & 1 \\
0 & \cdots & 0 & \lambda_i
\end{pmatrix}_{r_i \times r_i}\]
特征值$\lambda_1,\cdots,\lambda_k$可以重复，$r_i$为特征值$\lambda_i$的重数，在不考虑顺序时分解$A_1,A_2,\cdots,A_k$唯一。
\end{theorem}
\begin{proof}
取引理\ref{lemma:mlfj}中的$T=\lambda\mathbbm{I}-A$即证分解的存在性，下面寻找该分解。考虑$A$的特征多项式$p(\lambda)=\prod\limits_{i=1}^k(\lambda-\lambda_i)^{r_i}$，我们记$T_{\lambda_i}=\lambda_i\mathbbm{I}-A$，由引理\ref{lemma:mlsz}知有分解$X=\ker T_{\lambda_1}^{r_1} \oplus R(T_{\lambda_1}^{r_1})$，记$X_1=R(T_{\lambda_1}^{r_1})$，在$X_1$上$(A-\lambda_i\mathbbm{I})^{r_1}=T_{\lambda_1}^{r_1} \neq 0$，由 Cayley–Hamilton 定理[定理\ref{theorem:Cayley–Hamilton}]可知$p_2(A) \equiv p(A)/(A-\lambda_i\mathbbm{I})^{r_1}=0$为限制在$X_1$上的零矩阵，归纳得：
\[X=\bigoplus_{i=1}^k\ker T_{\lambda_i}^{r_i}=\bigoplus_{i=1}^k\ker(\lambda_i\mathbbm{I}-A)^{r_i}\]
\end{proof}
\newpage

如果要将上述有限维的结论推广到紧算子谱，对引理\ref{lemma:mlfj}只需用数学归纳法推广至可数维，这容易得到；对引理\ref{lemma:mlsz}则需证明零空间的分解在可数维中也成立，对此我们从定义零链长和像链长开始引理\ref{lemma:mlsz}的推广。
\begin{definition}[零链长，像链长]
设$X$是 Banach 空间，$T \in \mathscr{B}(X)$，若存在最小的$p \in \mathbb{N}_+$使得$\ker T^p=\ker T^{p+1}$，则称$p$为$T$的零链长；若存在最小的$q \in \mathbb{N}_+$使得$R(T^q)=R(T^{q+1})$，则称$q$为$T$的像链长。
\end{definition}
\begin{remark}
\quad 若$p,q<\infty$，则有$\ker T^p=\ker T^{p+1}=\cdots$，$R(T^q)=R(T^{q+1})=\cdots$，证明见引理\ref{lemma:mlsz}。
\end{remark}
\begin{lemma}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，则$T=\mathbbm{I}-A$的零链长等于像链长，均有限。
\end{lemma}
\begin{proof}
由$R(T) \subseteq X$可知$\{0\} \subseteq \ker T \subseteq \ker T^2 \subseteq \cdots$和$X \supseteq R(T) \supseteq R(T^2) \supseteq \cdots$。

采用反证法证明$q<+\infty$，如若不然则对$\forall k \in \mathbb{N}$都有$R(T^k) \supset R(T^{k+1})$，由$T^k=(\mathbbm{I}-A)^k=\mathbbm{I}-p(A)$，$p(A)$表示算子$A$的多项式，由定理\ref{theorem:szfh}可知$p(A)$也是紧算子。因此$R(T^k)$闭[定理\ref{theorem:Riesz-Fredholm}第二点]，由 Riesz 引理[定理\ref{Riesz}]，$\exists \, \{y_k\}$满足$y_k \in R(T^k)-R(T^{k+1})$且$\|y_k\|=1$，使得$d(y_k,R(T^{k+1}))>1/2$，但：
\[\|Ay_k-Ay_{k+l}\|=\|y_k-Ty_k-y_{k+l}+Ty_{k+l}\|\geq d(y_k,R(T^{k+1}))>\frac{1}{2}\]
其中$Ty_k \in R(T^{k+1})$，$y_{k+l} \in R(T^{k+l}) \subset R(T^{k+1})$，$Ty_{k+l} \in R(T^{k+l+1}) \subset R(T^{k+1})$，与$\{Ay_k\}$列紧矛盾。

上面证明了$q<+\infty$，因此$R(T^q)=R(T^{q+1})$。再证$p=q$，由 Riesz-Fredholm 定理[定理\ref{theorem:Riesz-Fredholm}第四点]可知：
\[\dim\ker T^q=\dim R[(T^q)^*]^{\perp}=\operatorname{codim}R(T^q)=\operatorname{codim}R(T^{q+1})=\dim\ker T^{q+1}\]
由$\dim\ker T^q<+\infty$ [定理\ref{theorem:Riesz-Fredholm}第四点]可知$\ker T^q=\ker T^{q+1}$，因此$q \geq p$，同理可证$q \leq p$，因此$p=q$。
\end{proof}

\begin{lemma}
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，则$T=\mathbbm{I}-A$满足$X=\ker T^p \oplus R(T^p)$，且$T_1=T|_{R(T^p)}$存在有界逆。
\end{lemma}
\begin{proof}
显然有$\ker T^p \cap R(T^p)=\{0\}$，则$\forall x \in X$，由于$p$是零链长和像链长，因此都有$T^px \in R(T^p)=R(T^{2p})$，也即$\exists \, u \in R(T^{2p})$满足$T^px=T^{2p}u$，令$y=T^pu$，则$z=x-y$满足$T^pz=T^px-T^py=T^px-T^{2p}u=0$，即$z \in \ker T^p$。综上所述，即对$\forall x \in X$都$\exists \, y \in R(T^p) \ , \ z \in \ker T^p$满足$x=y+z$，因此$X=\ker T^p \oplus R(T^p)$。

由于$R(T^p)=R(T^{p+1})$可知$T_1$是满射，同时若$y \in R(T^p)$使得$T_1y=0$，那$\exists \, x \in X$满足$y=T^px$使得$T_1y=T_1T^px=T^{p+1}x=0$，因此$x \in \ker T^{p+1}=\ker T^p$且$y=0$，由逆算子定理[定理\ref{theorem:nszdl}]可知$T_1$存在有界逆。
\end{proof}

\begin{theorem}[无穷维紧算子的 Jordan 分解定理]
设$X$是 Banach 空间，$A \in \mathscr{C}(X)$，对应$A$的非零谱即点谱$\lambda_i \in \sigma_p(A)$都存在对应$\lambda_i\mathbbm{I}-A$的零链长$p_i$，可以将空间$X$展开成：
\[X=\bigoplus_{i=1}^{\infty}\ker(\lambda_i\mathbbm{I}-A)^{p_i}\]
在上述展开空间中紧算子$A$有对应的 Jordan 标准型。
\end{theorem}

\section{希尔伯特空间的自共轭紧算子谱}\label{zgejszp}
更进一步的，继续优化算子和空间的性质，我们考虑希尔伯特空间上的自共轭紧算子谱。
\begin{definition}[共轭算子，自共轭算子，对称算子]
设$H$是 Hilbert 空间，若$A\in\mathscr{B}(H)$对$\forall x,y \in H$都$\exists \, A\in\mathscr{B}(H)$满足$(Ax,y)=(x,A^*y)$，则称$A^*$为$A$的共轭算子。若$A=A^*$则称$A$为自共轭算子。若$H$是实内积空间则称自共轭算子$A$为对称算子。
\end{definition}

\begin{example}
\quad $H=\mathbb{C}^n \ , \ A=\{a_{ij}\}_{n \times n}\in\mathscr{B}(H)$是自共轭算子，则有$(Ax,y)=x^{T}A^{T}\overline{y}=x^{T}\overline{Ay}=(x,Ay)$，取标准正交基则存在酉矩阵$P \in U(n)=\{p \in M_{n \times n}^{\mathbb{C}} \ | \ PP^{\dagger}=P^{\dagger}P=\mathbbm{I}_{n \times n}\}$，使得$P^{\dagger}AP=\text{diag}\{\lambda_1,\lambda_2,\cdots,\lambda_n\}$。
\end{example}

\begin{example}
\quad $H=L^2(\Omega) \ , \ \mathbb{K}=\mathbb{R} \ , \ K \in L^{\infty}(\Omega \times \Omega)$，定义算子$A$和$A^*$：
\[A: \ H \ \to \ H, \quad u(x) \ \mapsto \ \int_{\Omega}K(x,y)u(y)\dd{y} \quad ; \quad A^*: \ H \ \to \ H, \quad v(y) \ \mapsto \ \int_{\Omega}K(y,x)v(x)\dd{x}\]
则：
\[(Au,v)=\left(\int_{\Omega}K(x,y)u(y)\dd{y},v(x)\right)=\int_{\Omega}\int_{\Omega}K(x,y)u(y)v(x)\dd{y}\dd{x}=\left(u(y),\int_{\Omega}K(x,y)v(x)\dd{x}\right)=(u,A^*v)\]
当$K(x,y)=K(y,x)$时有$A=A^*$，此时$A$为自共轭算子。
\end{example}

\begin{example}
\quad 设$M \subset H$是 Hilbert 空间的子空间，$H=M \oplus M^{\perp}$，定义投影算子$p_M: \ H \to M \ , \ x=x_M+x_M^{\perp} \mapsto x_M$，则$(p_Mx,y)=(x_M,y_M+y_M^{\perp})=(x_M,y_M)=(x,p_My)$，即投影算子自共轭。
\end{example}

\begin{theorem}\label{theorem:zgeker}
设$H$是 Hilbert 空间，若$A\in\mathscr{B}(H)$是自共轭算子，则$(\ker A)^{\perp}=R(A)$。
\end{theorem}
\begin{proof}
因为$\ker A=\{x \in H \ | \ A(x)=0\}$，$R(A)=\{Ax \in AH \equiv H \ | \ \forall x \in H-\{0\}\}$，考虑共轭算子$A^*$，其值域$R(A^*)=\{A^*y \ | \ \forall y \in H-\{0\}\}=\{x \in H \ | \ A(x) \neq 0\}$，因此有$(\ker A)^{\perp} = R(A^*)$或$R(A)^{\perp}=\ker A^*$。由于$A$是自共轭算子，进而有$(\ker A)^{\perp}=R(A^*)=R(A)$。
\end{proof}

\begin{proposition}\label{proposition:zgesz}
设$H$是 Hilbert 空间，$A$为自共轭算子，则：
\begin{itemize}
\item 1. $\forall x \in H \ , \ (Ax,x)\in\mathbb{R}$当且仅当$A$为自共轭算子；
\item 2. $\|A\|=\sup\limits_{\|x\|=1}|(Ax,x)|$。
\end{itemize}
\end{proposition}
\begin{proof}
第一点必要性显然$(Ax,x) = (x,Ax) = \overline{(Ax,x)} \ \Rightarrow \ (Ax,x)\in\mathbb{R}$。下证充分性，考虑$(A(x+y),x+y)=(Ax,x)+(Ay,y)+(Ax,y)+(Ay,x) \in \mathbb{R}$，因此$\operatorname{Im}(Ax,y)+\operatorname{Im}(Ay,x)=0$，即$\operatorname{Im}(Ax,y)=\operatorname{Im}(x,Ay)$。考虑$(A(x+iy),x+iy)=(Ax,x)-(Ay,y)+i(Ax,y)+i(Ay,x) \in \mathbb{R}$，因此$\operatorname{Re}(Ax,y)+\operatorname{Re}(Ay,x)=0$，即$\operatorname{Re}(Ax,y)=\operatorname{Re}(Ay,x)=\operatorname{Re}(x,Ay)$，综上所述，$(Ax,y)=(x,Ay)$，充分性得证。

第二点，记$M = \sup\limits_{\|x\|=1}|(Ax,x)|$。由 Cauchy-Schwarz 不等式[定理\ref{theorem:Cauchy–Schwarz}]，$|(Ax,x)|\le \|A\|\cdot\|x\|=\|A\|$。取：
\[|(Ax,y)| = \frac{1}{4}\big| (A(x+y),x+y) - (A(x-y),x-y) \big| \le \frac{M}{4}\left(\|x+y\|^2+\|x-y\|^2\right)\]
取$\|x\|=\|y\|=1$，利用平行四边形法则，$\|x+y\|^2+\|x-y\|^2=2(\|x\|^2+\|y\|^2)=4$。因此$|(Ax,y)|\le M$对所有单位向量$x,y$成立。由算子范数的定义$\|A\|=\sup\limits_{\|x\|=\|y\|=1}|(Ax,y)|$，得$\|A\|\le M$。综上$\|A\|=M$。
\end{proof}

\begin{theorem}
设$H$是 Hilbert 空间，若$A\in\mathscr{B}(H)$是自共轭算子，则$\sigma(A) \subseteq \mathbb{R}$且$\sigma_r(A)=\varnothing$。
\end{theorem}
\begin{proof}
对$\forall \lambda=a+ib \in \mathbb{C}-\mathbb{R}$，其中$a,b \in \mathbb{R}$，则对$\forall x \in H$：
\[\begin{aligned}
\|(\lambda\mathbbm{I}-A)x\|^2&=\|(a\mathbbm{I}-A)x+ibx\|^2=\|(a\mathbbm{I}-A)x\|^2+\|bx\|^2-ib((a\mathbbm{I}-A)x,x)+ib(x,(a\mathbbm{I}-A)x)\\
&=\|(a\mathbbm{I}-A)x\|^2+\|bx\|^2-ib(x,(a\mathbbm{I}-A)x)+ib(x,(a\mathbbm{I}-A)x)=\|(a\mathbbm{I}-A)x\|^2+\|bx\|^2 \geq b^2\|x\|^2\\
\end{aligned}\]
上述公式第三个等号用到了性质\ref{proposition:zgesz}第一点，因此若$(\lambda\mathbbm{I}-A)x$可知$x=0$，即$\ker(\lambda\mathbbm{I}-A)=\{0\}$，由定理\ref{theorem:zgeker}可知$R(\lambda\mathbbm{I}-A)=(\ker(\lambda\mathbbm{I}-A))^{\perp}=H$，因此$\mathbb{C}-\mathbb{R} \subseteq \rho(A)$，因此$\sigma(A) \subseteq \mathbb{R}$。若$\lambda \in \sigma_r(A)$则$\ker(\lambda\mathbbm{I}-A)=\{0\}$且$\overline{R(\lambda\mathbbm{I}-A)} \neq H$。考虑$R(\lambda\mathbbm{I}-A)^{\perp}=\ker(\lambda\mathbbm{I}-A)=\{0\} \supset \overline{R(\lambda\mathbbm{I}-A)}^{\perp}$，因此$\overline{R(\lambda\mathbbm{I}-A)}=X$，即$\sigma_r(A)=\varnothing$。
\end{proof}
\begin{remark}
\quad 设$\lambda\neq\mu$是$A$的两个特征值，$x,y$分别为对应的特征向量，即$Ax=\lambda x$，$Ay=\mu y$。因此$\lambda(x,y) = (\lambda x,y) = (Ax,y) = (x,Ay) = (x,\mu y) = \mu(x,y)$，于是$(\lambda-\mu)(x,y)=0 \ \Rightarrow \ (x,y)=0$，即$x\perp y$，或$\ker(\lambda\mathbbm{I}-A) \perp \ker(\mu\mathbbm{I}-A)$。
\end{remark}

\begin{theorem}
设$H$是 Hilbert 空间，若$A\in\mathscr{B}(H)$是自共轭算子，$M$是$H$的闭子空间，则$A|_M$也是自共轭算子。
\end{theorem}
在 Hilbert 空间上，自共轭紧算子$A$的谱和算子结构将更楚。对比有穷维情形的对称阵$A$，它可以通过正交变換对角化，对角线上的元对应着$A$的本征值，而这些本征值双是$A$所对应的二次型$(Ax,x)$在单位球面$\|x\|=1$上的各个临界值。 所有这些性质将被推广到无穷维空间。我们从下列极值性质出发
\begin{theorem}
$H$是 Hilbert 空间，$A$是自共轭紧算子，则$\exists \, x_0 \in H$满足$\|x_0\|=1$且$Ax_0=\lambda x_0$，其中$\lambda\in\mathbb{R}$且$|\lambda|=\|A\|$。
\end{theorem}
\begin{proof}
不妨设$\lambda>0$，否则可以考虑$-A$算子。由性质\ref{proposition:zgesz}第二点可知存在$\lambda=\sup\limits_{\|x\|=1}(Ax,x)=\|A\|$，因此存在$\{x_n\} \subset S$使得$(Ax_n,x_n) \to \lambda$，由定理\ref{theorem:dwqrb}可知单位球$B$若闭，故存在$\{x_n\}$的弱收敛子列仍记为$\{x_n\}$满足$x_n \xrightarrow{w} x_0 \in B$，而$A$是紧算子，因此$A$全有界，进而$\|Ax_n-Ax_0\| \to 0$，故$(Ax_0,x_0)=\lambda$。只需证$x_0 \in S$。假设$\|x_0\|<1$，则$(Ax_0,x_0) \leq \|A\|\cdot\|x_0\|^2<\lambda$，矛盾。因此$x_0 \in S$。最后来验证$Ax_0=\lambda x_0$，由上确界可知$\lambda$是极大值。对$\forall y \in H$考察函数$\phi_y(t)$，假设$t$足够小，忽略其二阶项：
\[\phi_y(t)=\frac{(A(x_0+ty),x_0+ty)}{(x_0+ty,x_0+ty)}=\frac{(Ax_0,x_0)+2\operatorname{Re}[t(Ay,x_0)]}{(x_0,x_0)+2\operatorname{Re}[t(y,x_0)]}=\operatorname{Re}\left[\frac{(Ax_0,x_0)+2t(Ay,x_0)}{(x_0,x_0)+2t(y,x_0)}\right]\]
按照上面的结论可知$\phi_y(t)$在$t=0$处取极大值，$\phi'_y(0)=0$。对$\phi_y(t)$求$t$的形式导数：
\[\begin{aligned}
\dv{\phi_y}{t}&=\operatorname{Re}\left[\dv{t}\frac{(Ax_0,x_0)+2t(Ay,x_0)}{(x_0,x_0)+2t(y,x_0)}\right]=\operatorname{Re}\left[\frac{2(Ay,x_0)[(x_0,x_0)+2t(y,x_0)]-2(y,x_0)[(Ax_0,x_0)+2t(Ay,x_0)]}{((x_0,x_0)+2t(y,x_0))^2}\right]\\
&=2\operatorname{Re}\left[\frac{(Ay,x_0)(x_0,x_0)-(y,x_0)(Ax_0,x_0)}{((x_0,x_0)+2t(y,x_0))^2}\right]=\operatorname{Re}\left[\frac{2(Ay,x_0)-2\lambda(y,x_0)}{((x_0,x_0)+2t(y,x_0))^2}\right]=\operatorname{Re}\left[\frac{2(y,Ax_0-\lambda x_0)}{((x_0,x_0)+2t(y,x_0))^2}\right]
\end{aligned}\]
当$Ax_0=\lambda x_0$时即得$\phi'_y(0)=0$。
\end{proof}

\[x=\sum_{i}(x,e_i)e_i \quad , \quad Ax=\sum_{i}\lambda_i(x,e_i)e_i\]
\end{theorem}
